        function sortTable() {
            const table = document.getElementById("newpayabletable");
            const tbody = table.querySelector("tbody");

            const rows = Array.from(tbody.querySelectorAll("tr.charge"));
            const specialRow = tbody.querySelector("tr.special");

            const rowData = rows.map(row => {
                const syTerm = row.cells[0].textContent.trim();
                const description = row.cells[1].textContent.trim();
                const dateCharged = new Date(row.cells[2].textContent.trim());

                return { row, syTerm, description, dateCharged };
            });

            rowData.sort((a, b) => {

                var z = a.description.toLowerCase();

                if (!z.includes("payment")) {

                    if (a.syTerm > b.syTerm) return -1;
                    if (a.syTerm < b.syTerm) return 1;

                    if (a.description < b.description) return -1;
                    if (a.description > b.description) return 1;

                    if (a.dateCharged > b.dateCharged) return -1;
                    if (a.dateCharged < b.dateCharged) return 1;

                    return 0;

                }
            });

            rowData.forEach(({ row }) => tbody.appendChild(row));

            if (specialRow) {
                tbody.appendChild(specialRow);
            }
        }
